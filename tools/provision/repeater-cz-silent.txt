# Provisioning a repeater for the live CZ mesh that must ship SILENT.
#
#   tools/ble_cli.py <MAC> -f tools/provision/repeater-cz-silent.txt --per-command
#
# Order matters: `time` first, because a node whose RTC still says May 2024 has
# clients dropping its replies, and everything after that is easier to read in
# the log with a sane timestamp.
#
# What is NOT here, on purpose:
#   name / lat / lon / frequency / advert intervals / repeat
# Those come from the BUILD (see the deployment envs in platformio.local.ini).
# Baking them in means an `erase` on a mast rebuilds a correct, silent node
# instead of an upstream-default chatty one -- setting them here would only
# paper over a firmware that is wrong for the job.
#
# Run `verify-cz-silent.txt` afterwards. Do not ship a node you have not
# verified: on a mast there is no second chance.

# --- clock -------------------------------------------------------------
# The RTC is volatile. GPS re-syncs it after every reboot, which is why GPS is
# on even where the position is never published. If GPS already has a fix this
# returns "clock cannot go backwards" -- that is success, not an error.
# Substitute the real epoch; ble_cli does not expand shell.
# time <epoch>

# --- GPS: time yes, position not necessarily ---------------------------
# `duty`, not `on`: measured on this hardware, a continuously powered receiver
# costs 46 mA (paired A/B, 2026-08-05) -- more than the rest of the node's
# budget put together, and this node runs GPS purely to keep its volatile RTC
# set. Duty mode powers the receiver only around a clock sync, at the same
# 30-minute cadence `gps on` already re-syncs at, so the clock is no less
# accurate; it saves ~42-45 mA of the 46. If a fix never comes (blocked sky) it
# gives up after 5 minutes and retries in an hour instead of staying on.
gps duty
# `prefs` advertises the coordinates compiled into the firmware, NOT the live
# fix. That is what lets a node sit at a private address, keep accurate time and
# still show up in roughly the right place on the community maps.
# Use `share` only if publishing the exact spot is intended; `none` to publish
# no position at all.
gps advert prefs

# --- radio -------------------------------------------------------------
# 869.4-869.65 is ERC 70-03 band g3: 10 % duty. The firmware does not enforce
# duty on its own (airtime_factor defaults to 0 = 100 %) and caps at 9, so 9 is
# both the correct value here and the strictest the firmware can do.
set af 9
# Boosted RX gain stays ON despite costing ~0.7 mA: this node's whole job is
# hearing distant repeaters, and 1-3 dB of sensitivity is not for sale at that
# price. It is the one power knob deliberately left in the expensive position.
set radio.rxgain on

# --- power -------------------------------------------------------------
# The MCU sleeps between events (nRF52: WFE, wakes on any interrupt including
# the LoRa DIO1). Upstream leaves this OFF, and a repeater with it off runs the
# core at 64 MHz around the clock for no benefit. It was missing here, so every
# node provisioned from this file has been paying for it.
powersaving on
